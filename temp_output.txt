
        {/* SUGGESTED QUESTIONS */}
        <section>
          <h3 className="text-sm font-semibold text-primary mb-3 tracking-wide uppercase">
            Suggested OHRS Questions
          </h3>

          <div className="flex flex-wrap gap-2">
            {SUGGESTED_QUESTIONS.map((q) => (
              <button
                key={q}
                onClick={() => handleSuggestionClick(q)}
                className="text-sm px-4 py-2 rounded-full bg-white shadow-sm border border-primary/30 text-primary hover:bg-primary/10 transition"
              >
                {q}
              </button>
            ))}
          </div>
        </section>

        {/* CHAT MESSAGES */}
        {messages.map((m, idx) => {
          const citations =
            m.role === "assistant" ? extractCitations(m.content) : [];

          return (
            <div
              key={idx}
              className={`flex ${
                m.role === "user" ? "justify-end" : "justify-start"
              }`}
            >
              <div
                className={`max-w-[75%] px-4 py-3 rounded-2xl text-sm shadow-sm ${
                  m.role === "user"
                    ? "bg-blue-600 text-white rounded-br-md"
                    : "bg-white border border-surface border-l-4 border-primary/60 text-text rounded-bl-md"
                }`}
              >
                {m.role === "assistant" ? (
                  <div className="prose prose-sm max-w-full text-text leading-relaxed">
                    <ReactMarkdown
                      remarkPlugins={[remarkGfm]}
                      components={{
                        h1: ({ node, ...props }) => (
                          <h1 className="text-xl font-semibold text-primary mt-4 mb-2" {...props} />
                        ),
                        h2: ({ node, ...props }) => (
                          <h2 className="text-lg font-semibold text-primary mt-4 mb-2" {...props} />
                        ),
                        ul: ({ node, ...props }) => (
                          <ul className="list-disc list-outside ml-4 space-y-1" {...props} />
                        ),
                        ol: ({ node, ...props }) => (
                          <ol className="list-decimal list-outside ml-4 space-y-1" {...props} />
                        ),
                        p: ({ node, ...props }) => (
                          <p className="my-2 leading-[1.55]" {...props} />
                        ),
                        code: ({ node, ...props }) => (
                          <code className="bg-base px-1.5 py-0.5 rounded text-xs" {...props} />
                        ),
                        pre: ({ node, ...props }) => (
                          <pre className="bg-base p-3 rounded-lg text-xs overflow-auto" {...props} />
                        ),
                        table: ({ node, ...props }) => (
                          <table className="border-collapse border border-surface my-3 w-full text-xs" {...props} />
                        ),
                        th: ({ node, ...props }) => (
                          <th className="border border-surface px-2 py-1 bg-base/40 text-left" {...props} />
                        ),
                        td: ({ node, ...props }) => (
                          <td className="border border-surface px-2 py-1" {...props} />
                        ),
                      }}
                    >
                      {m.content}
                    </ReactMarkdown>
                  </div>
                ) : (
                  <div className="whitespace-pre-wrap leading-relaxed">{m.content}</div>
                )}

                {/* Sources Used */}
                {citations.length > 0 && (
                  <div className="mt-3 bg-base/70 border-l-2 border-primary/50 px-3 py-2 rounded text-xs text-text/80">
                    <div className="font-semibold text-primary/90 mb-1">Sources Used</div>
                    {citations.map((c) => (
                      <div key={c.id + c.page}>
                        Appendix <strong>{c.appendix}</strong>, Page <strong>{c.page}</strong>
                      </div>
                    ))}
                  </div>
                )}

                {/* CITATION TAGS */}
                {citations.length > 0 && (
                  <div className="flex mt-2 flex-wrap gap-2">
                    {citations.map((c) => (
                      <span
                        key={c.id + c.page}
                        onClick={() => setOpenCitation(c)}
                        className="px-2 py-1 text-xs bg-primary/10 text-primary rounded-md
                        border border-primary/20 cursor-pointer hover:bg-primary/20 transition"
                      >
                        {c.id} • Appendix {c.appendix} • Page {c.page}
                      </span>
                    ))}
                  </div>
                )}
              </div>
            </div>
          );
        })}

        {/* TYPING INDICATOR */}
        {loading && (
          <div className="flex justify-start">
            <div className="bg-white border border-surface px-4 py-3 rounded-2xl">
              <div className="flex gap-1">
                <span className="w-2 h-2 bg-text/40 rounded-full animate-bounce"></span>
                <span className="w-2 h-2 bg-text/40 rounded-full animate-bounce delay-150"></span>
                <span className="w-2 h-2 bg-text/40 rounded-full animate-bounce delay-300"></span>
              </div>
            </div>
          </div>
        )}

        <div ref={bottomRef} />
      </main>

      {/* INPUT BAR */}
      <form
        onSubmit={handleSubmit}
        className="w-full bg-white border-t border-surface p-4"
      >
        <div className="max-w-3xl mx-auto flex items-center gap-3">
          <input
            type="text"
            value={input}
            onChange={(e) => setInput(e.target.value)}
            placeholder="Message OHRS Assistant…"
            className="flex-1 px-4 py-3 rounded-xl border border-surface bg-base text-sm focus:ring-2 focus:ring-primary/30 outline-none"
          />

          <button
            type="submit"
            disabled={loading || !input.trim()}
            className="px-6 py-3 bg-primary text-white rounded-xl text-sm font-medium hover:bg-primary/90 disabled:bg-primary/40"
          >
            Send
          </button>
        </div>
      </form>

      {/* CITATION DRAWER */}
      {openCitation && (
        <div className="fixed inset-0 bg-black/20 backdrop-blur-sm flex justify-end z-50">
          <div className="w-[350px] bg-white shadow-xl p-5 border-l border-surface animate-slideIn">
            <div className="flex justify-between items-center mb-4">
              <h2 className="font-semibold text-primary">Citation Details</h2>
              <button
                onClick={() => setOpenCitation(null)}
                className="text-text/60 hover:text-text"
              >
                ✕
              </button>
            </div>

            <div className="text-sm text-text space-y-3">
              <div className="font-mono text-xs bg-base p-2 rounded border">
                {openCitation.id}
              </div>

              <p>
                <strong>Appendix:</strong> {openCitation.appendix}
              </p>
              <p>
                <strong>Page:</strong> {openCitation.page}
              </p>

              <div className="whitespace-pre-wrap text-sm leading-relaxed border border-surface rounded p-3 bg-base/60 max-h-[70vh] overflow-auto">
                {selectedPageText}
              </div>
            </div>
          </div>
        </div>
      )}

      {/* INTRO DRAWER */}
      {showIntroDrawer && (
        <div className="fixed inset-0 bg-black/20 backdrop-blur-sm flex justify-end z-50">
          <div className="w-[380px] bg-white shadow-xl p-6 border-l border-surface animate-slideIn space-y-5">
            <div className="flex justify-between items-center">
              <h2 className="text-xl font-semibold text-primary">
                Understanding OHRS
              </h2>
              <button
                onClick={() => setShowIntroDrawer(false)}
                className="text-text/60 hover:text-text"
              >
                ✕
              </button>
            </div>

            <section className="space-y-2">
              <h3 className="font-semibold text-sm text-primary">What is OHRS?</h3>
              <p className="text-sm text-text leading-relaxed">
                OHRS (Ontario Healthcare Reporting Standards) defines how
                organizations classify, count, and report financial, statistical,
                and functional centre activity to the Ministry of Health.
              </p>
              <ul className="list-disc list-inside text-sm text-text space-y-1">
                <li>Which functional centre to use</li>
                <li>What statistical accounts apply</li>
                <li>How workload & earned hours must be coded</li>
                <li>How to report bed occupancy, admissions, clinic visits</li>
              </ul>
            </section>

            <section className="space-y-2">
              <h3 className="font-semibold text-sm text-primary">
                What’s new in OHRS v12?
              </h3>
              <ul className="list-disc list-inside text-sm text-text space-y-1">
                <li>Updated functional centre groupings</li>
                <li>Revised statistical accounts</li>
                <li>Clarified LTC + community care workload rules</li>
                <li>Expanded reporting guidelines</li>
                <li>Improved structure across Appendix A–D</li>
              </ul>
            </section>

            <section className="space-y-2">
              <h3 className="font-semibold text-sm text-primary">
                How this tool helps
              </h3>
              <ul className="list-disc list-inside text-sm text-text space-y-1">
                <li>Answers backed only by OHRS Appendix A–D v12.0</li>
                <li>Every response includes exact citations</li>
                <li>Tap citations to view the exact OHRS page text</li>
                <li>Fully PHIPA-safe; runs only on local sources</li>
              </ul>
            </section>

            <button
              onClick={() => setShowIntroDrawer(false)}
              className="px-4 py-2 w-full bg-primary text-white rounded-lg hover:bg-primary/90"
            >
              Start using OHRS Assistant
            </button>
          </div>
        </div>
      )}
    </div>
  );
}
